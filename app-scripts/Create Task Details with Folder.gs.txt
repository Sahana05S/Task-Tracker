function createTaskDetailswithFolder(){
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('2_Tasks');
  
  // Get the active range list (handles discontinuous selections)
  var activeRangeList = sheet.getActiveRangeList().getRanges();
  
  // Loop through each individual range
  for (var r = 0; r < activeRangeList.length; r++) {
    var range = activeRangeList[r];

    // Loop through each row in the current range
    for (var i = 0; i < range.getNumRows(); i++) {
      var currentCell = range.getCell(i + 1, 1); // Assuming column 1 for task tag (adjust if needed)
      var currentRow = currentCell.getRow();

      // Check if the cell is within the allowed range (A6:A)
      if (currentRow >= 6 && currentCell.getColumn() === 1) {
        // Run the function to create the new task details page

         var columnJValue = sheet.getRange(currentRow, 10).getValue();
        var newSheet = CreateNewTaskDetailsPageForRow(currentCell); // Using the helper function for each cell

        if (newSheet) {
          // Set the newly created sheet as active
          SpreadsheetApp.getActiveSpreadsheet().setActiveSheet(newSheet);

          // Run the folder creation function with the new sheet as the active sheet
          createFolderFromSelection(columnJValue);
        }
      } else {
        // Notify the user that the selected cell is outside the allowed range
        SpreadsheetApp.getUi().alert('No processing done for cell ' + currentCell.getA1Notation() + '. It is outside the allowed range A6:A.');
      }
    }
  }

  SpreadsheetApp.getUi().alert('Task details and folders have been created for the valid selected cells.');
}
